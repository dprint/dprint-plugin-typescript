== should maintain the line breaks when the first element is on a new line ==
const t = [
    "testing", "testing",
    "testing", "testing", "testing"
];

[expect]
const t = [
    "testing", "testing",
    "testing", "testing", "testing",
];

== should maintain the line breaks when hanging ==
const t = ["node", "script",
    "-p", "./a/package.json",
    "-v", "1.339.0"];
const u = ["testing",
    "testing", "testing",
    "testing", "testing"];

[expect]
const t = ["node", "script",
    "-p", "./a/package.json",
    "-v", "1.339.0"];
const u = ["testing",
    "testing", "testing",
    "testing", "testing"];

== should fill the lines when hanging and there are no line breaks ==
const t = ["testing", "testing", "testing", "testing", "testing"];

[expect]
const t = ["testing", "testing",
    "testing", "testing", "testing"];

    "testing"];

[expect]

    "testing"];
