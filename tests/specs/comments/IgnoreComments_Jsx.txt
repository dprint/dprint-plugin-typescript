== should ignore element after ignore comment in an element ==
const t = (
    <Element>
        {/* dprint-ignore */}
        <Test      />
    </Element>
);

[expect]
const t = (
    <Element>
        {/* dprint-ignore */}
        <Test      />
    </Element>
);

    </>
);

    </>
);
