== should keep the blank lines between the members of a union type ==
export type K =
    // First
    | A.ONE
    | A.TWO

    // Second
    | A.THREE
    | A.FOUR;

[expect]
export type K =
    // First
    | A.ONE
    | A.TWO

    // Second
    | A.THREE
    | A.FOUR;

== should keep at most the max blank lines between the members ==
export type L =
    | A.ONE
    | A.TWO // trailing


    | A.THREE

    | A.FOUR;
export type O = A.ONE

    | A.TWO;

[expect]
export type L =
    | A.ONE
    | A.TWO // trailing

    | A.THREE

    | A.FOUR;
export type O =
    | A.ONE

    | A.TWO;

== should keep the blank lines between the members of an intersection type ==
export type N =
    & A.ONE

    /* Second */
    & A.THREE

    & { a: string;
     b: number }

    & A.FOUR;

[expect]
export type N =
    & A.ONE

    /* Second */
    & A.THREE

    & { a: string; b: number }

    & A.FOUR;
