== should add trailing comma after arrow expression body when call breaks across lines ==
const test = ["a","b","c"].map(value =>
    value == "a quite long paragraph with many words")

[expect]
const test = ["a", "b", "c",].map(value =>
    value == "a quite long paragraph with many words",
);

    someVeryLongCondition && anotherCondition,
);
