== should keep a jsdoc before a semicolon at the start of a line with the next statement ==
import { a } from "b"

/**
 * Initializes the app
 */
;(function initReact() {
  a()
})()

[expect]
import { a } from "b";

/**
 * Initializes the app
 */
(function initReact() {
    a();
})();
