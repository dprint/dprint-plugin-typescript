== should not count newlines in the input for the object siblings ==
const values = [{
  prop: "aaaaaaaaaa",
}, {
  other: "bbbbbbbbbb",
}];

[expect]
const values = [
  { prop: "aaaaaaaaaa" },
  { other: "bbbbbbbbbb" },
];

== should not count newlines in the input for type literals in a tuple ==
type T = [{
  prop: "aaaaaaaaaa";
}, {
  other: "bbbbbbbbbb";
}];

[expect]
type T = [
  { prop: "aaaaaaaaaa" },
  { other: "bbbbbbbbbb" },
];
