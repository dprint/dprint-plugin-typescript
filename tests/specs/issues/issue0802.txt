== should format exported declarations with a line break after the keyword ==
export type
T = keyof X;
export abstract
class X {}
export interface
I {}
export namespace
Foo {}
export module
Bar {}
export enum
E {}

[expect]
export type T = keyof X;
export abstract class X {}
export interface I {}
export namespace Foo {}
export module Bar {}
export enum E {}

== should format exported declarations with comments ==
export /* a */ namespace /* b */ Foo {
    export namespace Bar {}
}

[expect]
export /* a */ namespace /* b */ Foo {
    export namespace Bar {}
}

== should format an exported abstract class with a decorator ==
@dec export abstract class Y {}

[expect]
@dec
export abstract class Y {}
