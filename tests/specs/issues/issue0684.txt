    doSomething()
    // eslint-disable-next-line react-hooks/exhaustive-deps

    () => doSomething(),

== should preserve multiple comments between argument and comma ==
foo(
    a
    // c1
    /* c2 */
    // c3
    , b);

[expect]
foo(
    a,
    // c1
    /* c2 */
    // c3
    b,
);
