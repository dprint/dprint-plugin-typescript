== should format graphql in graphql`...` ==
const q = graphql`query   User {   id   name   }`;
[expect]
const q = graphql`query User { id name }`;

== should format graphql in gql`...` ==
const q = gql`mutation   Foo {   bar   }`;
[expect]
const q = gql`mutation Foo { bar }`;

== should format graphql in graphql.experimental`...` ==
graphql.experimental`mutation   MarkRead($input: Bar)   {   markRead(data: $input)  {  id  } }`;
[expect]
graphql.experimental`mutation MarkRead($input: Bar) { markRead(data: $input) { id } }`;

== should not format graphql in other members of graphql ==
graphql.other`query   { users { name } }`;
[expect]
graphql.other`query   { users { name } }`;

== should not format a template literal that's an argument of graphql ==
graphql(`query   { users { name } }`);
[expect]
graphql(`query   { users { name } }`);
