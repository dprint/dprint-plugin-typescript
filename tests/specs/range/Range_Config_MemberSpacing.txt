== should format a class member with a blank line between members ==
class   A   {
    a  =  1;
    [|b  =  2;|]
    c  =  3;
}

[expect]
class   A   {
    a  =  1;
[|    b = 2;|]
    c  =  3;
}

== should format an interface member with a blank line between members ==
interface   A   {
    a :  string;

    [|b :  string;|]

    c :  string;
}

[expect]
interface   A   {
    a :  string;

[|    b: string;|]

    c :  string;
}
