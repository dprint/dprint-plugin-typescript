== should format a variable statement in a namespace ==
namespace   N   {
    const  a  =  1;
    [|let   x  =  1,   y  =  2;|]
    const  b  =  2;
}

[expect]
namespace   N   {
    const  a  =  1;
[|    let x = 1, y = 2;|]
    const  b  =  2;
}
