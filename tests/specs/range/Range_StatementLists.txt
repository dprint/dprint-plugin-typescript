== should format the touched statements in a function declaration body ==
const  x  =  1;
function   test( )   {
    const  a  =  1;
    [|call( a );|]
}

[expect]
const  x  =  1;
function   test( )   {
    const  a  =  1;
[|    call(a);|]
}
