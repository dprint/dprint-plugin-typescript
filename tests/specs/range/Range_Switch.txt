== should format the last statement in a case without the ones before it ==
switch   ( x )   {
    case   1:
        const  a  =  1;
            [|break;|]
    default:
        const  b  =  1;
}

[expect]
switch   ( x )   {
    case   1:
        const  a  =  1;
[|        break;|]
    default:
        const  b  =  1;
}

== should format a multi-line last statement in a case without the ones before it ==
switch   ( x )   {
    case   1:
        const  a  =  1;
        [|if   ( a )   {
 call( );
        }|]
    default:
        const  b  =  1;
}

[expect]
switch   ( x )   {
    case   1:
        const  a  =  1;
[|        if (a) {
            call();|]
        }
    default:
        const  b  =  1;
}

== should format a statement in a case ending in a non-ascii identifier ==
switch   ( a )   {
    case   1:
        foo(  [|1|]  )
        x  =  café
}

[expect]
switch   ( a )   {
    case   1:
[|        foo(1);|]
        x  =  café
}

== should format the comments the range touches after the last statement in a case ==
switch   ( a )   {
    case   1:
        x( );
        [|y( );
              // c|]
    case   2:
        z( );
}

[expect]

    case   1:
        x( );
[|        y();
        // c|]
    case   2:
        z( );
}
