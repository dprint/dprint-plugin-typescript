== should format an import declaration in a declared module ==
declare   module   "m"   {
    const  a :  string;
    [|import   x ,  {  y  }   from   "x";|]
    const  b :  string;
}

[expect]
declare   module   "m"   {
    const  a :  string;
[|    import x, { y } from "x";|]
    const  b :  string;
}
