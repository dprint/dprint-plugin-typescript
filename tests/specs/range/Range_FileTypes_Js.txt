== should format a with statement in a switch case ==
function   test( )   {
    switch   ( x )   {
        case   1:
            var   a  =  1;
            [|with   (  obj  )   {
                call( a );
            }|]
    }
}

[expect]
function   test( )   {
    switch   ( x )   {
        case   1:
            var   a  =  1;
[|            with (obj) {
                call(a);|]
            }
    }
}
