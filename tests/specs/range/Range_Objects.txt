== should format a key value property ==
const   a  =  {
    b :  1,
    [|c :  2,|]
    d :  3,
};

[expect]
const   a  =  {
    b :  1,
[|    c: 2,|]
    d :  3,
};

== should format a string key property ==
const   a  =  {

    d :  3,
};

[expect]
const   a  =  {
    b :  1,
[|    "c": 2,|]
    d :  3,
};

    d :  3,

== should format the statement when the object is on a single line ==
const   x  =  1;
const   a  =  {   b :  1,   [|c :  2|]   };

[expect]
const   x  =  1;
[|const a = { b: 1, c: 2 };|]

== should format the statement when the object will be on a single line ==

    [|d :  2|] };

[expect]
const   x  =  1;
[|const b = { c: 1, d: 2 };|]

== should format the touched properties with comma first style ==

    , [|c :  2|]

};

[expect]
[|const a = {
    b: 1,
    c: 2,
    d: 3,|]
};
