== should format the member holding a keyword type ==
namespace   A   {
    type   X  =  1;
    type   Y  =  [|string|];
    type   Z  =  2;
}

[expect]
namespace   A   {
    type   X  =  1;
[|    type Y = string;|]
    type   Z  =  2;
}
