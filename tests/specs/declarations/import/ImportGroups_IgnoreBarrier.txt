== an ignored import is a barrier and stays where it is ==
// dprint-ignore
import { a } from "react";
import { fs } from "node:fs";

[expect]
// dprint-ignore
import { a } from "react";

import { fs } from "node:fs";

== imports on either side of the barrier group independently ==
import { z } from "zzz";
import { fs } from "node:fs";
// dprint-ignore
import { pinned } from "pinned";
import { c } from "./c";
import { path } from "node:path";

[expect]
import { fs } from "node:fs";

import { z } from "zzz";
// dprint-ignore
import { pinned } from "pinned";

import { path } from "node:path";

import { c } from "./c";

== the ignored import keeps its own formatting ==
// dprint-ignore
import {a,   b} from "react";
import { fs } from "node:fs";

[expect]
// dprint-ignore
import {a,   b} from "react";

import { fs } from "node:fs";

== dprint-ignore-start counts as an ignore comment, so it pins the import after it ==
// dprint-ignore-start
import { a } from "react";
import { fs } from "node:fs";
// dprint-ignore-end

[expect]
// dprint-ignore-start
import { a } from "react";

import { fs } from "node:fs";
// dprint-ignore-end
