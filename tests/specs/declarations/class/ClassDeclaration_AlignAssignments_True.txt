== should align the values of the properties ==
class Test {
a = 1;
longPropertyName = "value";
static b = true;
private c?: number = 3;
#d = 4;
accessor e = 5;
method() {
return 42;
}
}

[expect]
class Test {
  a                  = 1;
  longPropertyName   = "value";
  static b           = true;
  private c?: number = 3;
  #d                 = 4;
  accessor e         = 5;
  method() {
    return 42;
  }
}
