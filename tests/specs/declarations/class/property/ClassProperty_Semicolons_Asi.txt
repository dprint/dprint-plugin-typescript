== should not use a semi-colon when false ==
class Test {
    prop: string;
}

[expect]
class Test {
    prop: string
}

== should not remove semi on property before computed ==
class Test {
    readonly a = "prop";
    ["b"] = 2;
    ["c"]() {
        return 0;
    }
}

[expect]
class Test {
    readonly a = "prop";
    ["b"] = 2;
    ["c"]() {
        return 0
    }
}

== should not remove semi on property before computed after an empty member ==

    ;
    [k] = 2;
}

[expect]
class Test {
    a = 1;

    [k] = 2
}
