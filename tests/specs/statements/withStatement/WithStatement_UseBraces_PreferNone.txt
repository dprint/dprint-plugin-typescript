== should not use braces when only one line ==
with (true) {
    a;
}

[expect]
with (true)
    a;

== should use braces when condition is multiple lines ==
with (true && false && true && false && true)
    a;

[expect]
with (true && false && true && false
    && true)
{
    a;
}
