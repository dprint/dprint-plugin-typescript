~~ switchStatement.spaceAround: true ~~
== should format with space around the switch param ==
switch (foo) {
}

[expect]
switch ( foo ) {
}

== should not add spaces at the end or start of a line around the switch param ==
switch (someVeryLongConditionIdentifierName <= anotherVeryLongConditionIdentifierName && yetAnotherVeryLongConditionName) {
}

[expect]
switch (
    someVeryLongConditionIdentifierName <= anotherVeryLongConditionIdentifierName && yetAnotherVeryLongConditionName
) {
}
